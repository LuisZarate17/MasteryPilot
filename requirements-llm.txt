# MasteryPilot — local inference and RAG stack.
#
# Not needed to install the site, log in, browse the dashboard, or run the test
# suite. Every package here is imported lazily, the first time a chat starts
# (see ensure_llm_initialized() in code/llmsite/tutor/views/chat.py). Install
# it when you want the tutor to actually answer.
#
# Expect roughly 3-5 GB on disk, most of it torch.
#
# CPU-only (the default, works everywhere):
#
#     pip install -r requirements-llm.txt
#
# NVIDIA GPU — install torch from PyTorch's index FIRST, then this file. Note
# --index-url, not --extra-index-url: it applies to torch alone and does not
# mix package indexes for everything else.
#
#     pip install torch==2.7.1 --index-url https://download.pytorch.org/whl/cu128
#     pip install -r requirements-llm.txt
#
# cu128, not cu118. This file said cu118 until 2026-08-13, which installs
# without complaint and then fails at the first kernel launch on any Blackwell
# card (RTX 50-series, sm_120) with "no kernel image available for device" —
# the cu118 build carries no sm_120 cubins. Verified on an RTX 5070 Ti: the
# cu128 build of the same pinned 2.7.1 reports
# sm_50/60/61/70/75/80/86/90/100/120 and runs. PEP 440 treats torch==2.7.1 as
# satisfied by 2.7.1+cu128, so the CUDA build survives the second command.
# Older cards are unaffected either way; cu128 covers sm_50 upward.

# RAG retrieval (code/llmsite/rag.py)
numpy==2.5.1
PyPDF2==3.0.1
faiss-cpu==1.15.0
sentence-transformers==5.7.0

# .docx ingestion. Required, not optional: the dashboard accepts .docx uploads,
# and without this rag.py's extractor returns nothing — so the file would
# upload, appear in the curriculum list, and never be indexed. Silently
# ignoring a teacher's document is worse than refusing it.
python-docx==1.2.0

# Local model adapters (code/llmsite/languagemodel.py).
# transformers is held below 5.0 deliberately: the adapters were written and
# demoed against the 4.x API and have not been revalidated against 5.x.
torch==2.7.1
transformers==4.57.6
accelerate==1.14.0
bitsandbytes==0.50.0

# Required by Mistral's tokenizer, which is SentencePiece-based and is
# converted through protobuf on first load. Without these,
# LLM_MODULE=mistral fails at AutoTokenizer.from_pretrained with
# "requires the protobuf library but it was not found" — before any weight is
# read, so the failure looks like a broken model rather than a missing
# dependency. Qwen and LLaMA ship fast tokenizers and do not need them, which
# is why this went unnoticed while qwen was the only path anyone ran.
protobuf==7.35.1
sentencepiece==0.2.2
